-- Place one ShopFlow order in a single transaction.
-- orders.total is set from the order items before COMMIT.
\set ON_ERROR_STOP on
\timing on
Timing is on.
\echo '=== Stock and orders before the transaction ==='
=== Stock and orders before the transaction ===
SELECT id, name, stock
FROM products
WHERE name = 'Notebook';

----+----------+-------
  7 | Notebook |   150
(1 row)

Time: 10.393 ms
SELECT COUNT(*) AS orders, MAX(id) AS max_order_id
FROM orders;
 orders | max_order_id 
--------+--------------
  20005 |        20005
(1 row)

Time: 13.461 ms
BEGIN;
BEGIN
Time: 0.897 ms
INSERT INTO orders (customer_id, status, total)
SELECT id, 'confirmed', 0
FROM customers
WHERE email = 'alice@example.com'
RETURNING id \gset order_
INSERT 0 1
Time: 11.650 ms
INSERT INTO order_items (order_id, product_id, quantity, price)
SELECT :order_id, p.id, 2, p.price
FROM products p
WHERE p.name = 'Notebook';
INSERT 0 1
Time: 5.690 ms
UPDATE products
SET stock = stock - 2
WHERE name = 'Notebook';
UPDATE 1
Time: 2.180 ms
INSERT INTO inventory_logs (product_id, "change", reason)
SELECT id, -2, 'Portfolio 3 order fulfilment'
FROM products
WHERE name = 'Notebook';
INSERT 0 1
Time: 2.946 ms
UPDATE orders AS o
SET total = (
    SELECT SUM(quantity * price)
    FROM order_items
    WHERE order_id = o.id
)
WHERE o.id = :order_id;
UPDATE 1
Time: 7.637 ms
\echo '=== Order total and stock before COMMIT ==='
=== Order total and stock before COMMIT ===
SELECT
    o.id,
    o.status,
    o.total AS recorded_total,
    SUM(oi.quantity * oi.price) AS calculated_total
FROM orders AS o
JOIN order_items AS oi ON oi.order_id = o.id
WHERE o.id = :order_id
GROUP BY o.id, o.status, o.total;
  id   |  status   | recorded_total | calculated_total

(1 row)

Time: 3.569 ms
SELECT name, stock

Time: 0.626 ms
COMMIT;
